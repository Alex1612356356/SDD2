Sursa și Implementarea

Programul este implementat integral în C++ și utilizează fișierul date.in pentru stocarea informațiilor inițiale.

Fiecare înregistrare din fișier include titlul jocului, anul lansării, dificultatea și ratingul. Codul sursă este optimizat pentru a rula pe orice compilator C++ modern.

Codul Sursă Complet

Mai jos este prezentată implementarea completă a programului Game Library în C++. 

#include <iostream>
#include <fstream>
#include <string>
using namespace std;
ifstream fin("date.in");
struct nod {
    string titluJoc;
    int anLansare;
    int nivelDificultate;
    float ratingJucatori;
    nod* urm;
    nod* ante;
};
void creare(nod*& prim, nod*& ultim)
{
    int n;
    fin >> n;
    fin.ignore();
    prim = ultim = NULL;
    for (int i = 1; i <= n; i++)
    {
        nod* p = new nod;
        getline(fin, p->titluJoc);
        fin >> p->anLansare;
        fin >> p->nivelDificultate;
        fin >> p->ratingJucatori;
        fin.ignore();
        p->urm = NULL;
        p->ante = ultim;
        if (prim == NULL)
            prim = p;
        else
            ultim->urm = p;
        ultim = p;
    }
}
void parcurgere(nod* prim)
{
    if (prim == NULL)
    {
        cout << "Lista este vida\n";
        return;
    }
    while (prim != NULL)
    {
        cout << prim->titluJoc << " - "
            << prim->anLansare << " - dificultate: "
            << prim->nivelDificultate << " - rating: "
            << prim->ratingJucatori << "\n";
        prim = prim->urm;
    }
}
void inserare(nod*& prim, nod*& ultim)
{
    string titlu;
    cout << "Titlul jocului dupa care inseram: ";
    cin.ignore();
    getline(cin, titlu);
    nod* p = prim;
    while (p != NULL && p->titluJoc != titlu)
        p = p->urm;
    if (p == NULL)
    {
        cout << "Jocul nu exista\n";
        return;
    }
    nod* q = new nod;
    cout << "Titlu nou: ";
    getline(cin, q->titluJoc);
    cout << "An lansare: ";
    cin >> q->anLansare;
    cout << "Nivel dificultate: ";
    cin >> q->nivelDificultate;
    cout << "Rating: ";
    cin >> q->ratingJucatori;
    q->urm = p->urm;
    q->ante = p;
    if (p->urm != NULL)
        p->urm->ante = q;
    else
        ultim = q;
    p->urm = q;
}
void stergere(nod*& prim, nod*& ultim)
{
    string titlu;
    cout << "Titlul jocului de sters: ";
    cin.ignore();
    getline(cin, titlu);
    nod* p = prim;
    while (p != NULL && p->titluJoc != titlu)
        p = p->urm;
    if (p == NULL)
    {
        cout << "Jocul nu exista\n";
        return;
    }
    if (p == prim)
    {
        prim = prim->urm;
        if (prim != NULL)
            prim->ante = NULL;
        else
            ultim = NULL;
    }
    else if (p == ultim)
    {
        ultim = ultim->ante;
        ultim->urm = NULL;
    }
    else
    {
        p->ante->urm = p->urm;
        p->urm->ante = p->ante;
    }
    delete p;
}
void ordonareRating(nod* prim)
{
    if (prim == NULL) return;
    bool modificat;
    do {
        modificat = false;
        nod* p = prim;
        while (p->urm != NULL)
        {
            if (p->ratingJucatori < p->urm->ratingJucatori)
            {
                swap(p->titluJoc, p->urm->titluJoc);
                swap(p->anLansare, p->urm->anLansare);
                swap(p->nivelDificultate, p->urm->nivelDificultate);
                swap(p->ratingJucatori, p->urm->ratingJucatori);
                modificat = true;
            }
            p = p->urm;
        }
    } while (modificat);
}
int numarJocuri(nod* prim)
{
    int nr = 0;
    while (prim != NULL)
    {
        nr++;
        prim = prim->urm;
    }
    return nr;
}
void ratingMaxim(nod* prim)
{
    if (prim == NULL) return;
    nod* max = prim;
    prim = prim->urm;
    while (prim != NULL)
    {
        if (prim->ratingJucatori > max->ratingJucatori)
            max = prim;
        prim = prim->urm;
    }
    cout << "Joc cu rating maxim: "
        << max->titluJoc << " - "
        << max->ratingJucatori << "\n";
}
void ratingMinim(nod* prim)
{
    if (prim == NULL) return;
    nod* min = prim;
    prim = prim->urm;
    while (prim != NULL)
    {
        if (prim->ratingJucatori < min->ratingJucatori)
            min = prim;
        prim = prim->urm;
    }
    cout << "Joc cu rating minim: "
        << min->titluJoc << " - "
        << min->ratingJucatori << "\n";
}
float medieRating(nod* prim)
{
    if (prim == NULL) return 0;
    float suma = 0;
    int nr = 0;
    while (prim != NULL)
    {
        suma += prim->ratingJucatori;
        nr++;
        prim = prim->urm;
    }
    return suma / nr;
}
void jocuriDupaAn(nod* prim, int an)
{
    while (prim != NULL)
    {
        if (prim->anLansare > an)
            cout << prim->titluJoc << " - " << prim->anLansare << "\n";
        prim = prim->urm;
    }
}
void eliberare(nod*& prim)
{
    nod* p;
    while (prim != NULL)
    {
        p = prim;
        prim = prim->urm;
        delete p;
    }
}
int main()
{
    nod* prim;
    nod* ultim;
    creare(prim, ultim);
    int optiune;
    do {
        cout << "\n      MENIU \n";
        cout << "1. Afisare lista\n";        cout << "2. Inserare joc\n";
        cout << "3. Stergere joc\n";
        cout << "4. Ordonare dupa rating\n";
        cout << "5. Numar jocuri\n";
        cout << "6. Rating maxim\n";
        cout << "7. Rating minim\n";
        cout << "8. Media ratingurilor\n";
        cout << "9. Jocuri dupa un an\n";
        cout << "0. Iesire\n";
        cout << "Optiunea: ";
        cin >> optiune;
        switch (optiune)
        {
        case 1:
            parcurgere(prim);
            break;
        case 2:
            inserare(prim, ultim);
            break;
        case 3:
            stergere(prim, ultim);
            break;
        case 4:
            ordonareRating(prim);
            cout << "Lista a fost ordonata\n";
            break;
        case 5:
            cout << "Numar total jocuri: "
                << numarJocuri(prim) << "\n";
            break;
        case 6:
            ratingMaxim(prim);
            break;
        case 7:
            ratingMinim(prim);
            break;
        case 8:
            cout << "Media ratingurilor: "
                << medieRating(prim) << "\n";
            break;
        case 9:
        {
            int an;
            cout << "Introdu anul: ";
            cin >> an;
            jocuriDupaAn(prim, an);
            break;
        }
        case 0:
            break;
        default:
            cout << "Optiune invalida\n";
        }
    } while (optiune != 0);
    eliberare(prim);
    return 0;
}
